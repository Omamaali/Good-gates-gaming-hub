<script>

/* ================= SCORE ================= */
let points = localStorage.getItem("p") || 0;
document.getElementById("points").innerText = points;

function add(x){
points = parseInt(points) + x;
localStorage.setItem("p", points);
document.getElementById("points").innerText = points;
Yuya.react(points);
}

/* ================= YUYA GLOBAL ================= */
const Yuya = {
talk: function(msg){
document.getElementById("box").style.display="block";
document.getElementById("talk").innerText = msg;
},

toggle: function(){
let b=document.getElementById("box");
b.style.display = (b.style.display=="block")?"none":"block";
},

react: function(p){
if(p > 300){
this.talk("🏆 Legendary Player!");
}else if(p > 150){
this.talk("🔥 Great progress!");
}else{
this.talk("Keep exploring 🌍");
}
}
};

setTimeout(()=>Yuya.talk("Welcome to GoodGate Global 🌍"),1200);

/* ================= QUIZ ================= */

<button onclick="checkQuiz(false)">No</button>

}

}

}

/* ================= HIERO QUEST ================= */
let word="";

function hiero(){
clear();

let words=["SUN","KING","WATER","LOVE","POWER"];
word = words[Math.floor(Math.random()*words.length)];

let map={
S:"𓋴",H:"𓉔",A:"𓄿",N:"𓈖",
K:"𓎡",I:"𓇋",R:"𓂋",
W:"𓅱",O:"𓅱",E:"𓇌"
};

let h="";
for(let c of word){
h+=map[c]||"𓂀";
}

document.getElementById("hiero").innerHTML=`
<h3>🏺 Hiero Quest</h3>
<p>Guess the word:</p>
<div style="font-size:40px">${h}</div>
<input id="g">
<button onclick="check()">Check</button>
<p id="res"></p>
`;
}

function check(){
let g=document.getElementById("g").value.toUpperCase();

if(g==word){
add(150);
Yuya.talk("🏺 Ancient Master!");
document.getElementById("res").innerText="Correct!";
}else{
Yuya.talk("Try again explorer 🌍");
}
}
